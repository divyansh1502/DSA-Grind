```java
/*
===============================================================================
                    JAVA COLLECTION INTERFACE
===============================================================================

Package:
    java.util.Collection

Collection is an INTERFACE.

It is one of the root interfaces of the Java Collection Framework.

Basic hierarchy:

                    Collection<E>
                         |
          --------------------------------
          |              |               |
         List            Set            Queue
          |              |               |
     ArrayList        HashSet      PriorityQueue
     LinkedList       LinkedHashSet
                      TreeSet

IMPORTANT:
    Map is NOT a child of Collection.

    Map stores:
        Key -> Value

    Collection stores:
        Elements

===============================================================================
*/


/*
===============================================================================
                    COLLECTION vs COLLECTIONS
===============================================================================

Collection
-----------
    -> Interface
    -> Represents a group of objects
    -> Part of java.util
    -> Provides common methods for collections
    -> Implemented by List, Set and Queue interfaces

Example:

    Collection<Integer> c = new ArrayList<>();


Collections
-----------
    -> Utility class
    -> Contains static utility methods
    -> Used to perform operations on collections

Examples:

    Collections.sort(list);
    Collections.reverse(list);
    Collections.shuffle(list);
    Collections.max(list);
    Collections.min(list);
    Collections.frequency(list, 10);


MEMORY TRICK:

    Collection  = INTERFACE / CONTRACT

    Collections = UTILITY CLASS / TOOLBOX

===============================================================================
*/


import java.util.*;


/*
===============================================================================
                IMPORTANT METHODS OF COLLECTION INTERFACE
===============================================================================

1. add()
2. addAll()
3. remove()
4. removeAll()
5. retainAll()
6. clear()
7. contains()
8. containsAll()
9. size()
10. isEmpty()
11. toArray()
12. toArray(T[] array)
13. iterator()
14. removeIf()
15. stream()
16. parallelStream()
17. spliterator()
18. equals()
19. hashCode()

===============================================================================
*/


/*
===============================================================================
1. add(E e)
===============================================================================

Purpose:
    Adds one element to the collection.

Syntax:

    boolean add(E e);

Return:
    true  -> collection changed
    false -> collection did not change

===============================================================================
*/


public class CollectionAddExample {

    public static void main(String[] args) {

        Collection<Integer> numbers = new ArrayList<>();

        numbers.add(10);
        numbers.add(20);
        numbers.add(30);

        System.out.println(numbers);

        /*
            Output:

            [10, 20, 30]
        */
    }
}


/*
===============================================================================
                    add() WITH ARRAYLIST
===============================================================================

ArrayList allows:
    -> Duplicate elements
    -> Insertion order
    -> null values

Example:
*/


ArrayList<Integer> list = new ArrayList<>();

list.add(10);
list.add(20);
list.add(20);
list.add(30);

System.out.println(list);

// [10, 20, 20, 30]


/*
===============================================================================
                    add() WITH LINKEDLIST
===============================================================================

LinkedList also allows:
    -> Duplicate elements
    -> Insertion order
    -> null values

*/


LinkedList<Integer> linkedList = new LinkedList<>();

linkedList.add(10);
linkedList.add(20);
linkedList.add(20);

System.out.println(linkedList);

// [10, 20, 20]


/*
===============================================================================
                    add() WITH HASHSET
===============================================================================

HashSet:
    -> Does NOT allow duplicate elements
    -> Does NOT guarantee insertion order

*/


HashSet<Integer> set = new HashSet<>();

set.add(10);
set.add(20);
set.add(10);

System.out.println(set);

/*
    Output can be:

    [20, 10]

    The important point:
        10 appears only once.
*/


/*
===============================================================================
                    TIME COMPLEXITY OF add()
===============================================================================

ArrayList:
    Usually O(1) amortized when adding at the end.

LinkedList:
    O(1) for adding at the end.

HashSet:
    Average O(1)

===============================================================================
*/


/*
===============================================================================
2. addAll(Collection<? extends E> c)
===============================================================================

Purpose:
    Adds ALL elements of another collection.

Syntax:

    boolean addAll(Collection<? extends E> c);

===============================================================================
*/


ArrayList<Integer> first = new ArrayList<>();

first.add(10);
first.add(20);

ArrayList<Integer> second = new ArrayList<>();

second.add(30);
second.add(40);

first.addAll(second);

System.out.println(first);

// [10, 20, 30, 40]


/*
===============================================================================
                    addAll() WITH HASHSET
===============================================================================
*/


HashSet<Integer> numbersSet = new HashSet<>();

numbersSet.add(10);

numbersSet.addAll(Arrays.asList(20, 30, 40));

System.out.println(numbersSet);

// [10, 20, 30, 40]


/*
===============================================================================
IMPORTANT:

Collection<? extends E>

This means:

    Collection containing E
    OR
    Collection containing subclass of E

Example:

    Collection<Number> numbers = new ArrayList<>();

    List<Integer> integers = Arrays.asList(10, 20);

    numbers.addAll(integers);

This is possible because Integer extends Number.

===============================================================================
*/


/*
===============================================================================
3. remove(Object o)
===============================================================================

Purpose:
    Removes ONE occurrence of the specified object.

Syntax:

    boolean remove(Object o);

Returns:

    true  -> element removed
    false -> element not found

===============================================================================
*/


ArrayList<String> names =
        new ArrayList<>(Arrays.asList("A", "B", "C"));

names.remove("B");

System.out.println(names);

// [A, C]


/*
===============================================================================
VERY IMPORTANT INTERVIEW POINT

List<Integer> has TWO remove() methods:

    remove(int index)
    remove(Object o)

Example:
*/


ArrayList<Integer> nums =
        new ArrayList<>(Arrays.asList(10, 20, 30));

nums.remove(1);

/*
    Here 1 is int.

    Therefore:

        remove(int index)

    Result:

        [10, 30]
*/


nums.remove(Integer.valueOf(30));

/*
    Here we explicitly create Integer.

    Therefore:

        remove(Object)

    Result:

        [10]
*/


/*
===============================================================================
4. removeAll(Collection<?> c)
===============================================================================

Purpose:

    Removes ALL elements from the current collection
    that are also present in the given collection.

Think:

    removeAll = SUBTRACTION

===============================================================================
*/


ArrayList<Integer> numbers =
        new ArrayList<>(Arrays.asList(1, 2, 2, 3, 4));

numbers.removeAll(Arrays.asList(2, 4));

System.out.println(numbers);

// [1, 3]


/*
===============================================================================
HashSet example
===============================================================================
*/


HashSet<Integer> hashSet =
        new HashSet<>(Arrays.asList(1, 2, 3, 4));

hashSet.removeAll(Arrays.asList(2, 4));

System.out.println(hashSet);

// [1, 3]



/*
===============================================================================
5. retainAll(Collection<?> c)
===============================================================================

Purpose:

    Keeps ONLY elements that are present in both collections.

Think:

    retainAll = KEEP COMMON ELEMENTS

===============================================================================
*/


ArrayList<Integer> values =
        new ArrayList<>(Arrays.asList(1, 2, 3, 4));

values.retainAll(Arrays.asList(2, 4, 6));

System.out.println(values);

// [2, 4]


/*
===============================================================================
MEMORY TRICK

removeAll():

    Remove common elements

    A = [1,2,3,4]
    B = [2,4]

    A.removeAll(B)

    Result:
        [1,3]


retainAll():

    Keep common elements

    A = [1,2,3,4]
    B = [2,4]

    A.retainAll(B)

    Result:
        [2,4]

===============================================================================
*/


/*
===============================================================================
6. clear()
===============================================================================

Purpose:

    Removes ALL elements.

Syntax:

    void clear();

===============================================================================
*/


ArrayList<String> data =
        new ArrayList<>(Arrays.asList("Java", "SQL", "Spring"));

data.clear();

System.out.println(data);

// []


System.out.println(data.size());

// 0


System.out.println(data.isEmpty());

// true


/*
===============================================================================
7. contains(Object o)
===============================================================================

Purpose:

    Checks whether an element exists.

Return:

    true
    false

===============================================================================
*/


ArrayList<String> languages =
        new ArrayList<>(Arrays.asList("Java", "Python", "C++"));

System.out.println(languages.contains("Java"));

// true

System.out.println(languages.contains("JavaScript"));

// false


/*
===============================================================================
8. containsAll(Collection<?> c)
===============================================================================

Purpose:

    Checks whether the current collection contains
    ALL elements of another collection.

===============================================================================
*/


ArrayList<Integer> numbersList =
        new ArrayList<>(Arrays.asList(10, 20, 30, 40));

System.out.println(
        numbersList.containsAll(Arrays.asList(10, 30))
);

// true


System.out.println(
        numbersList.containsAll(Arrays.asList(10, 50))
);

// false


/*
IMPORTANT:

containsAll() does NOT mean:

    "Both collections are equal."

It means:

    "Does this collection contain every element
     requested by the other collection?"

===============================================================================
*/


/*
===============================================================================
9. size()
===============================================================================

Purpose:

    Returns number of elements.

Return type:

    int

Syntax:

    int size();

===============================================================================
*/


ArrayList<Integer> list2 =
        new ArrayList<>(Arrays.asList(10, 20, 30));

System.out.println(list2.size());

// 3


/*
Typical complexity:

    ArrayList  -> O(1)
    LinkedList -> O(1)
    HashSet    -> O(1)

===============================================================================
*/


/*
===============================================================================
10. isEmpty()
===============================================================================

Purpose:

    Checks whether collection contains ZERO elements.

Return:

    true  -> empty
    false -> contains elements

===============================================================================
*/


ArrayList<Integer> emptyList = new ArrayList<>();

System.out.println(emptyList.isEmpty());

// true


emptyList.add(10);

System.out.println(emptyList.isEmpty());

// false


/*
Relationship:

    collection.isEmpty()

is logically equivalent to:

    collection.size() == 0

But isEmpty() directly expresses the intention.

===============================================================================
*/


/*
===============================================================================
11. toArray()
===============================================================================

Purpose:

    Converts collection into an Object[] array.

Return:

    Object[]

===============================================================================
*/


ArrayList<String> fruits =
        new ArrayList<>(Arrays.asList("Apple", "Mango", "Banana"));

Object[] array = fruits.toArray();

System.out.println(Arrays.toString(array));

// [Apple, Mango, Banana]


/*
===============================================================================
12. toArray(T[] array)
===============================================================================

Purpose:

    Converts collection into a specific type of array.

Example:
*/


String[] fruitArray =
        fruits.toArray(new String[0]);

System.out.println(Arrays.toString(fruitArray));

// [Apple, Mango, Banana]


/*
Difference:

    toArray()
        -> Object[]

    toArray(new String[0])
        -> String[]

===============================================================================
*/


/*
===============================================================================
13. iterator()
===============================================================================

Iterator is used to traverse collection elements ONE BY ONE.

Syntax:

    Iterator<E> iterator();

Important Iterator methods:

    hasNext()
    next()
    remove()

===============================================================================
*/


ArrayList<Integer> marks =
        new ArrayList<>(Arrays.asList(70, 80, 90));

Iterator<Integer> iterator = marks.iterator();

while (iterator.hasNext()) {

    Integer mark = iterator.next();

    System.out.println(mark);
}

/*
Output:

70
80
90
*/


/*
===============================================================================
WHY ITERATOR?

Instead of directly accessing the internal structure of a collection,
Iterator provides a standard way to traverse it.

Different collections have different internal structures:

    ArrayList  -> dynamic array
    LinkedList -> linked nodes
    HashSet    -> hash table

But Iterator gives a common traversal mechanism.

===============================================================================
*/


/*
===============================================================================
ITERATOR AND SAFE REMOVAL
===============================================================================
*/


ArrayList<Integer> nums2 =
        new ArrayList<>(Arrays.asList(10, 20, 30, 40));

Iterator<Integer> it = nums2.iterator();

while (it.hasNext()) {

    Integer value = it.next();

    if (value == 20) {
        it.remove();
    }
}

System.out.println(nums2);

// [10, 30, 40]


/*
IMPORTANT:

During iteration, avoid doing:

    nums2.remove(value);

because structural modification directly through the collection
can cause ConcurrentModificationException.

Use:

    iterator.remove();

when removing the current element through an Iterator.
*/


/*
===============================================================================
14. removeIf(Predicate<? super E> filter)
===============================================================================

Introduced as a default method in Collection.

Purpose:

    Removes elements that satisfy a condition.

===============================================================================
*/


ArrayList<Integer> evenNumbers =
        new ArrayList<>(Arrays.asList(1, 2, 3, 4, 5, 6));

evenNumbers.removeIf(x -> x % 2 == 0);

System.out.println(evenNumbers);

// [1, 3, 5]


/*
Meaning:

    x -> x % 2 == 0

means:

    "If x is even, remove it."

===============================================================================
*/


/*
===============================================================================
15. stream()
===============================================================================

Returns a sequential Stream.

Used for:

    filtering
    mapping
    sorting
    counting
    collecting
    etc.

===============================================================================
*/


ArrayList<Integer> numbers3 =
        new ArrayList<>(Arrays.asList(10, 20, 30, 40, 50));

long count =
        numbers3.stream()
               .filter(x -> x > 20)
               .count();

System.out.println(count);

// 3


/*
===============================================================================
16. parallelStream()
===============================================================================

Creates a parallel Stream.

It can divide stream processing across multiple threads.

===============================================================================
*/


numbers3.parallelStream()
        .forEach(System.out::println);


/*
IMPORTANT:

Parallel processing does NOT automatically mean faster.

For small collections or simple operations,
parallelStream() may provide no benefit or even add overhead.

===============================================================================
*/


/*
===============================================================================
17. spliterator()
===============================================================================

Spliterator =

    Split + Iterator

It can:

    -> Traverse elements
    -> Split elements into parts
    -> Support parallel processing

===============================================================================
*/


ArrayList<Integer> dataList =
        new ArrayList<>(Arrays.asList(10, 20, 30, 40));

Spliterator<Integer> spliterator =
        dataList.spliterator();

spliterator.forEachRemaining(System.out::println);


/*
===============================================================================
18. equals(Object o)
===============================================================================

Collection inherits equals() from Object.

But specific collection types define their own equality rules.

For example:

List equality:
    -> Elements must be equal
    -> Same order

===============================================================================
*/


List<Integer> listA =
        Arrays.asList(10, 20, 30);

List<Integer> listB =
        Arrays.asList(10, 20, 30);

System.out.println(listA.equals(listB));

// true


List<Integer> listC =
        Arrays.asList(30, 20, 10);

System.out.println(listA.equals(listC));

// false

/*
Because order is different.

===============================================================================
*/


/*
===============================================================================
19. hashCode()
===============================================================================

Every Java object inherits hashCode() from Object.

Collection implementations may provide hash codes according
to their equality contract.

IMPORTANT:

    If two objects are equal according to equals(),

    they MUST have the same hashCode().

But:

    Same hashCode()
        does NOT guarantee
    objects are equal.

===============================================================================
*/


/*
===============================================================================
             ARRAYLIST vs LINKEDLIST vs HASHSET
===============================================================================

                    ArrayList       LinkedList       HashSet
                    ---------       ----------       -------
Structure           Dynamic Array   Doubly Linked    Hash Table

Duplicates          YES             YES              NO

Insertion Order     YES             YES              NO guarantee

Index Access         Fast            Slow             No index

get(index)           O(1)            O(n)             Not available

contains()           O(n)            O(n)             Average O(1)

remove(Object)       O(n)            O(n)             Average O(1)

add(end)             Amortized O(1)  O(1)             Average O(1)

size()               O(1)            O(1)             O(1)

isEmpty()            O(1)            O(1)             O(1)

===============================================================================
*/


/*
===============================================================================
             SAME COLLECTION METHOD, DIFFERENT IMPLEMENTATION
===============================================================================

The Collection interface defines WHAT should be done.

The implementation decides HOW it should be done.

Example:

    Collection<Integer> c;

    c = new ArrayList<>();

    c.contains(10);

Here:

    Collection
        -> provides the contract

    ArrayList
        -> provides the implementation

===============================================================================

Another example:

    Collection<Integer> c;

    c = new LinkedList<>();

    c.contains(10);

The method call is the same:

    contains()

But the internal implementation is different.

===============================================================================
*/


/*
===============================================================================
              PROGRAMMING TO INTERFACE
===============================================================================

Recommended:

    List<Integer> list = new ArrayList<>();

instead of:

    ArrayList<Integer> list = new ArrayList<>();

Why?

Because List is an interface.

Later we can change:

    list = new LinkedList<>();

without changing code that only depends on List methods.

Similarly:

    Collection<Integer> c = new ArrayList<>();

can later become:

    Collection<Integer> c = new HashSet<>();

provided we only need Collection-level behavior.

===============================================================================
*/


/*
===============================================================================
                COMPLETE COLLECTION CHEAT SHEET
===============================================================================

ADDING
------

    add()
        -> Add one element

    addAll()
        -> Add multiple elements


REMOVING
--------

    remove()
        -> Remove one matching element

    removeAll()
        -> Remove elements found in another collection

    retainAll()
        -> Keep elements found in another collection

    clear()
        -> Remove everything


SEARCHING
---------

    contains()
        -> Does one element exist?

    containsAll()
        -> Do all requested elements exist?


STATE
-----

    size()
        -> Number of elements

    isEmpty()
        -> Is size zero?


CONVERSION
----------

    toArray()
        -> Object[]

    toArray(T[])
        -> Typed array


TRAVERSAL
---------

    iterator()
        -> Traverse one by one

    spliterator()
        -> Traverse + split


MODERN JAVA
-----------

    removeIf()
        -> Remove according to condition

    stream()
        -> Sequential Stream

    parallelStream()
        -> Parallel Stream


OBJECT METHODS
--------------

    equals()
        -> Logical equality according to collection contract

    hashCode()
        -> Hash code according to equality contract

===============================================================================
*/

ONE-LINE SUMMARY:

===============================================================================
```
